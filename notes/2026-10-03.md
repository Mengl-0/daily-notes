题目1：分割字符串的最大得分
![alt text](../images/2026-10-03_001.png)
``` Go
func maxScore(s string) int{
    n := len(s)
    if n < 2{
        return 0
    }
    score := 0
    if score == '0'{
        score = 1
    }else {
        score = -1
    }
    diff := 0
    maxDiff :=  -1<<31
    pre1 := 0
    for i :=0;i< n-1;i++{
        if s[i] == '0'{
            diff++
        }else{
            diff--
            pre1++
        }
        if diff > maxDiff{
            maxDiff = diff
        }
    }
    total := pre1
    if s[n-1] == '1'{
        total++
    }
    return maxDiff + total
}
```
```python
class Solution:
    def maxScore(self,s: str) -> int:
        total1 = s.count('1')
        left0 = 0
        max_val = 0
        n = len(s)
        for i in range(n-1):
            c = s[i]
            if c == '0'{
                left0 += 1
            }
            left1 = i+1 -left0
            right1 = total1 - left1
            score = left0 + right1
            max_val = max(max_val, score)
        return max_val
```

```Go
func peakIndexInMountainArray(arr []int) int{
    left := 0
    right := len(arr) - 1
    for left < right{
        mid = (left + right) / 2
        if arr[mid] < arr[mid + 1]{
            left ++
        }else{
            right = mid
        }
    }
    return left
}
```
``` python
class Solution:
    def peakIndexInMountainArray(self, arr:List[int]) -> int:
        left = 0
        right = len(arr) -1
        while left < right:
            mid = (left+right) / 2
            if arr[mid] < arr[mid+1]:
                left += 1
            else:
                right = mid
        return left
```
