题目1：下降路径和

``` Go
func minFallingPathSum(matrix [][]int) int {
    n := len(matrix)
    dp := make([][]int,n)
    for i:=0;i<n;i++{
        dp[i] = make([]int,n)
    }
    for i :=0;i<n;i++{
        dp[0][i] = matrix[0][i]
    }
    for i:=1;i<n;i++{
        for j:=0;j<n;j++{
            if j==0{
                dp[i][j] = min(dp[i-1][j],dp[i-1][j+1]) + matrix[i][j]
            }else if j==n-1{
                dp[i][j] = min(dp[i-1][j-1],dp[i-1][j]) + matrix[i][j]
            }else{
                dp[i][j] = min(min(dp[i-1][j-1],dp[i-1][j]),dp[i-1][j+1]) + matrix[i][j]
            }
        }
    }
    minVal := dp[n-1][0]
    for i :=1;i<n;i++{
        if minVal > dp[n-1][i]{
            minVal = dp[n-1][i]
        }
    }
    return  minVal
}
```
``` python
class Solution:
    def minFallingPathSum(self, matrix: list[list[int]]) -> int:
        n = len(matrix)
        dp = [[0]*n for _ in range(n)]
        for i in range(n):
            dp[0][i] = matrix[0][i]
        for i in range(1,n):
            for j in range(n):
                if j==0:
                    dp[i][j] = min(dp[i-1][j],dp[i-1][j+1]) + matrix[i][j]
                elif j==n-1:
                    dp[i][j] = min(dp[i-1][j],dp[i-1][j-1]) + matrix[i][j]
                else:
                    dp[i][j] = min(min(dp[i-1][j-1],dp[i-1][j]),dp[i-1][j+1]) + matrix[i][j]
        return min(dp[-1])
```

### 收获
通过下降路径最小和的练习，掌握多方向转移的二维动态规划解法，学会区分矩阵边界的特殊处理；理解滚动数组优化 DP 空间的原理，同时意识到编写转移方程不能主观简化逻辑，避免遗漏候选值产生错误。对比之前刷的不同路径、三角形最小路径和，理清矩阵 DP 题不同移动规则带来的写法差异，巩固了动态规划的解题套路。
