func minPathSum(grid [][]int) int {
    m := len(grid)
    n := len(grid[0])
    dp := make([][]int, n)
    for i in range m{
        dp[i] = make([]int,n)
    }
    for i := range m{
        if i == 0{
            dp[i][0] = grid[i][0]
        }else{
            dp[i][0] = dp[i-1][0] + grid[i][0]
        }
    }
    for j := range n{
        if j == 0{
            dp[0][j] = grid[0][j]
        }else{
            dp[0][j] = dp[0][j-1] + grid[0][j]
        }
    }
    for i:= 1;i<m;i++{
        for j:= 1;j<n;j++{
            dp[i][j] = min(dp[i-1][j],dp[i][j-1]) + grid[i][j]
        }
    }
    return dp[i-1][j-1]
}
```

func minimunTotal(triangle [][]int) int{
    m := len(triangle)
    dp := make([][]int ,m)
    for i := 0;i<m;i++{
        dp[i] = make([]int,i+1)
    }
    dp[0][0] = triangle[0][0]
    for i :=1;i<m;i++{
        dp[i][0] = dp[i-1][0] + triangle[i][0]
        dp[i][i] = dp[i-1][i-1] + triangle[i][i]
        for j := 1;j<i;j++{
            dp[i][j] = min(dp[i-1][j-1],dp[i-1][j]) + triangle[i][j]
        }
    }
    minVal := dp[m-1][0]
    for i:=1;i<m;i++{
        if minVal > dp[m-1][i]{
            minVal = dp[m-1][i]
        }
    }
    return minVal
}
```
``` python
class Solution:
    def minimunTotal(self, triangle:List[List[int]]) -> int:
        m = len(triangle)
        dp = [[0]*(i+1) for i in range(m)]
        dp[0][0] = triangle[0][0]
        for i in range(1,m):
            dp[i][0] = dp[i-1][0] + triangle[i][0]
            dp[i][i] = dp[i-1][i-1] + triangle[i][i]
            for j in range(1,i):
                dp[i][j] = min(dp[i-1][j-1],dp[i-1][j]) + triangle[i][j]
        return min(dp[m-1])
```
