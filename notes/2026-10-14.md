题目1：最长回文子序列
![alt text](../images/2026-10-14_002.png)
```Go
func longestPalindromeSubseq(s string) int {
    n := len(s)
    if n <=1{
        return n
    }
    dp := make([][]int,n)
    for i:=0;i<n;i++{
        dp[i] = make([]int,n)
    }
    for i:=0;i<n;i++{
        dp[i][i] = 1
    }
    for lenl := 2;lenl<=n;lenl++{
        for i:=0;i+lenl-1<n;i++{
            j:= i+lenl-1
            if s[j]==s[i]{
                if lenl==2{
                    dp[i][j] = 2
                }else{
                    dp[i][j] = dp[i+1][j-1] + 2
                }
            }else{
                dp[i][j] = max(dp[i+1][j],dp[i][j-1])
            }
        }
    }
    return dp[0][n-1]
}
```
```python
class Solution:
    def longestPalindromeSubseq(self, s: str) -> int:
        n = len(s)
        if n <=1:
            return n
        dp = [[0]*n for _ in range(n)]
        for i in range(n):
            dp[i][i] = 1
        for lenL in range(2,n+1):
            for i in range(n-lenL+1):
                j = i+lenL-1
                if s[i] == s[j]:
                    if lenL == 2:
                        dp[i][j] = 2
                    else:
                        dp[i][j] = dp[i+1][j-1] + 2
                else:
                    dp[i][j] = max(dp[i+1][j],dp[i][j-1])
            i +=1
        return dp[0][-1]
```

func minDistance(word1 string,word2 string) int{
    m := len(word1)
    n := len(word2)
    dp := make([][]int,m+1)
    for i:=0;i<=m;i++{
        dp[i] = make([]int,n+1)
    }
    for i:=0;i<=n;i++{
        dp[0][i] = i
    }
    for j:=0;j<=m;j++{
        dp[j][0] = j
    }
    for i:=1;i<=m;i++{
        for j:=1;j<=n;j++{
            if word1[i-1] == word2[j-1]{
                dp[i][j] = dp[i-1][j-1]
            }else{
                dp[i][j] = min(dp[i-1][j-1],dp[i-1][j],dp[i][j-1]) + 1
            }
        }
    }
    return dp[m][n]
}

### 收获
1. **DP 状态**：`dp[i][j]` = word1 前 i 个字符转为 word2 前 j 个字符的最小操作数。
   - 字符相等：`dp[i][j] = dp[i-1][j-1]`，无需操作
   - 字符不等：取替换、删除、插入三者最小值 + 1
   - 边界：`dp[i][0]=i`（全部删除），`dp[0][j]=j`（全部插入）
2. **实操方法**：动手手动推演 DP 表格，直观验证状态转移，排查逻辑错误。
3. **进阶优化**：二维 DP 可以压缩一维数组，空间复杂度从 O (mn) 降为 O (min (m,n))，需要临时变量保存左上角的值。
